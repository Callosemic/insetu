from insetu.core.sdk import InSetuExtension

SCHEMA = {
    "{{ext_name}}_items": {
        "id": "TEXT PRIMARY KEY",
        "name": "TEXT NOT NULL"
    }
}
SETTINGS = [
    {"id": "example_setting", "label": "Example Setting", "type": "text", "scope": "workspace", "default": ""}
]

{{ext_name}}_bp = InSetuExtension(
    '{{ext_name}}', 
    __name__,
    title="{{cap_name}}",
    description="A newly scaffolded extension.",
    schema=SCHEMA,
    settings_schema=SETTINGS
)
__depends__ = []
@{{ext_name}}_bp.route('list', methods=['GET'], docstring="[sync] Lists all stored items for this extension.")
def api_list(ctx):
    """Lists all stored items for this extension."""
    items = ctx.db.get_all("{{ext_name}}_items")
    return {"items": items}
